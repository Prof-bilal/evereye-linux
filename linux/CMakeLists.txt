cmake_minimum_required(VERSION 3.21)
project(EverEye VERSION 0.4.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "EverEye requires Linux D-Bus and process supervision facilities")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets DBus Concurrent Network)
if(BUILD_TESTING)
    find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
endif()
set(EVEREYE_SOURCES src/core.cpp src/platform.cpp src/guardian.cpp src/service.cpp src/window.cpp src/jobs.cpp)
add_executable(evereye src/main.cpp src/gateway.cpp ${EVEREYE_SOURCES})
target_link_libraries(evereye PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::DBus Qt6::Concurrent Qt6::Network rt util)
target_compile_options(evereye PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
    add_executable(insomnia-tests tests/test_insomnia.cpp ${EVEREYE_SOURCES})
    target_include_directories(insomnia-tests PRIVATE src)
    target_link_libraries(insomnia-tests PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::DBus Qt6::Concurrent Qt6::Test rt util)
    add_test(NAME regression COMMAND insomnia-tests)
    set_tests_properties(regression PROPERTIES TIMEOUT 150 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=;QT_STYLE_OVERRIDE=Fusion")
    add_executable(insomnia-fake-services tests/fake_services.cpp)
    target_link_libraries(insomnia-fake-services PRIVATE Qt6::Core Qt6::DBus)
    add_executable(insomnia-fake-services-legacy tests/fake_services.cpp)
    target_compile_definitions(insomnia-fake-services-legacy PRIVATE EVEREYE_LEGACY_PROFILES)
    target_link_libraries(insomnia-fake-services-legacy PRIVATE Qt6::Core Qt6::DBus)
    find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
    find_program(DBUS_RUN_SESSION dbus-run-session)
    set(DBUS_RUN_SESSION_VAR "${DBUS_RUN_SESSION}")
    if(DBUS_RUN_SESSION_VAR)
        add_test(NAME gateway COMMAND ${DBUS_RUN_SESSION_VAR} -- ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test-gateway.py)
        set_tests_properties(gateway PROPERTIES TIMEOUT 120 ENVIRONMENT "EVEREYE_BINARY=$<TARGET_FILE:evereye>")
    endif()
    if(DBUS_RUN_SESSION)
        foreach(api current legacy)
            if(api STREQUAL "legacy")
                set(fixture insomnia-fake-services-legacy)
            else()
                set(fixture insomnia-fake-services)
            endif()
            add_test(NAME private-bus-${api} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/private-bus-run.py $<TARGET_FILE:evereye> $<TARGET_FILE:${fixture}> --profile-api ${api})
            set_tests_properties(private-bus-${api} PROPERTIES TIMEOUT 150)
        endforeach()
    else()
        message(WARNING "dbus-run-session missing: private-bus E2E tests unavailable")
    endif()
endif()
install(TARGETS evereye RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/io.github.ProfBilal.EverEye.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/io.github.ProfBilal.EverEye.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES assets/io.github.ProfBilal.EverEye.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME evereye.svg)
install(FILES packaging/io.github.ProfBilal.EverEye.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/evereye)
option(EVEREYE_INSTALL_USER_SERVICE "Install the optional systemd user unit" ON)
set(EVEREYE_USER_UNIT_DIR "lib/systemd/user" CACHE STRING "systemd user unit path relative to prefix")
if(EVEREYE_INSTALL_USER_SERVICE)
    file(READ packaging/systemd/evereye.service unit)
    string(REPLACE "ExecStart=/usr/bin/" "ExecStart=${CMAKE_INSTALL_FULL_BINDIR}/" unit "${unit}")
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/evereye.service" CONTENT "${unit}")
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/evereye.service" DESTINATION ${EVEREYE_USER_UNIT_DIR})
endif()
option(EVEREYE_INSTALL_GATEWAY_UNIT "Install the opt-in remote gateway user unit" OFF)
if(EVEREYE_INSTALL_GATEWAY_UNIT)
    file(READ packaging/systemd/evereye-gateway.service gateway_unit)
    string(REPLACE "ExecStart=/usr/bin/" "ExecStart=${CMAKE_INSTALL_FULL_BINDIR}/" gateway_unit "${gateway_unit}")
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/evereye-gateway.service" CONTENT "${gateway_unit}")
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/evereye-gateway.service" DESTINATION ${EVEREYE_USER_UNIT_DIR})
endif()
option(EVEREYE_INSTALL_WEB "Install the built mobile PWA for the gateway to serve" ON)
if(EVEREYE_INSTALL_WEB AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../mobile/out/index.html")
    install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../mobile/out/" DESTINATION ${CMAKE_INSTALL_DATADIR}/evereye/web)
endif()
set(CPACK_PACKAGE_NAME evereye)
set(CPACK_PACKAGE_VENDOR "EverEye contributors")
set(CPACK_PACKAGE_CONTACT "EverEye contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Timed awake sessions with battery safeguards")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "EverEye contributors")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "dbus, upower")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "power-profiles-daemon")
set(CPACK_RPM_PACKAGE_LICENSE MIT)
set(CPACK_RPM_PACKAGE_AUTOREQ ON)
# `upower` pulls the distro's D-Bus service package. The daemon is named
# `dbus` on Fedora/RHEL and `dbus-1` on openSUSE, so do not encode a
# distribution-specific name in the shared RPM metadata.
set(CPACK_RPM_PACKAGE_REQUIRES "upower")
# openSUSE's `filesystem` package owns /usr/share/licenses. Keep EverEye's
# license files, but do not claim ownership of that shared parent directory.
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION "/usr/share/licenses")
set(CPACK_GENERATOR TGZ)
include(CPack)
